// ВЫСТАВИТЬ НА РЫНОК if (target.classList.contains('btn-list-item')) { let idx = parseInt(target.getAttribute('data-index')); let item = state.inventory[idx]; let inputPrice = prompt(`За сколько хотите продать номер ${item.phone}? (Вы получите 95% от суммы, 5% комиссия)`, Math.floor(item.price * 1.5)); if (!inputPrice || isNaN(inputPrice)) return; let listingPrice = parseInt(inputPrice); if (listingPrice < 100) { showToast("Минимальная цена: 100 ₽"); return; } if (window.db && window.fbRef && window.fbSet) { // Отправляем на глобальный рынок let newRef = window.fbRef(window.db, 'market/' + Date.now() + '_' + myId); // ВАЖНО: используем .then() и .catch() для обработки результата window.fbSet(newRef, { phone: item.phone, price: listingPrice, category: item.category, sellerId: myId, sellerName: myName, timestamp: Date.now() }).then(() => { // Если запись в базу прошла УСПЕШНО, только тогда удаляем из рюкзака state.inventory.splice(idx, 1); updateUI(); renderInventory(); showToast(`✅ Лот выставлен на рынок за ${listingPrice.toLocaleString('ru-RU')} ₽`); // Если рынок открыт, обновляем его if (getEl('global-market-screen').style.display === 'flex') { openGlobalMarket(); } }).catch((error) => { // Если база данных отклонила запись, выводим реальную ошибку console.error("Ошибка рынка Firebase:", error); showToast(`❌ Ошибка базы: ${error.message}`); }); } else { showToast("❌ Ошибка подключения к рынку."); } }
